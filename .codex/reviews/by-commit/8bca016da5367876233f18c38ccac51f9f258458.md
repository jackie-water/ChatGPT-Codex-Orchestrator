    46	function trustedHelper(file) {
    47	  if (typeof file !== "string" || !path.isAbsolute(file)) return null;
    48	  const uid = typeof process.getuid === "function" ? process.getuid() : null;
    49	  const trustedOwner = st => uid === null || st.uid === uid || st.uid === 0;
    50	  const validateChain = start => {
    51	    const normalized = path.normalize(start);
    52	    let current = path.parse(normalized).root;
    53	    for (const part of normalized.slice(current.length).split(path.sep).filter(Boolean)) {
    54	      current = path.join(current, part);
    55	      let st;
    56	      try { st = fs.lstatSync(current); } catch { return false; }
    57	      if (!trustedOwner(st) || (st.isDirectory() && (st.mode & 0o022) !== 0)) return false;
    58	      if (st.isSymbolicLink()) {
    59	        let target;
    60	        try { target = fs.realpathSync(current); } catch { return false; }
    61	        if (!validateChain(target)) return false;
    62	      } else if (current !== normalized && !st.isDirectory()) return false;
    63	    }
    64	    return true;
    65	  };
    66	  const validateOriginalAncestors = original => {
    67	    const root = path.parse(original).root;
    68	    let current = root;
    69	    for (const part of original.slice(root.length).split(path.sep).filter(Boolean)) {
    70	      if (part === ".") continue;
    71	      const candidate = part === ".." ? path.dirname(current) : path.join(current, part);
    72	      let st;
    73	      try { st = fs.lstatSync(candidate); } catch { return false; }
    74	      if (!trustedOwner(st) || (st.isDirectory() && (st.mode & 0o022) !== 0)) return false;
    75	      current = candidate;
    76	    }
    77	    return true;
    78	  };
    79	  let resolved;
    80	  try { resolved = fs.realpathSync(file); } catch { return null; }
    81	  if (!validateOriginalAncestors(file) || !validateChain(file) || !validateChain(resolved)) return null;
    82	  let stat;
    83	  try { stat = fs.lstatSync(resolved); } catch { return null; }
    84	  if (!stat.isFile() || stat.nlink !== 1 || (stat.mode & 0o111) === 0 || (stat.mode & 0o022) !== 0 || !trustedOwner(stat)) return null;
    85	  return resolved;
    86	}
